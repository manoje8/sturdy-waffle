3) peterson number

Idea:
input = 145

1 = 1! = 1
4 = 4! = 24
5 = 5! = 120

total = 145
input === total

->
Get an input as n
Use while-loop, condition will be n is greater than 0
Initialize variable as remainder in ouside the while loop
In while-loop, remainder = n % 10 to get the remainder value.
use the for-loop inside while-loop, Initialize i= 1, condition = i <=n, iterate = i++
initiaze variable as sum = 1, just outside the for-loop
Inside for-loop, sum = sum * i
And then Initialize varaible as result ouside while loop, to store the sum value.
Ouside for-loop, result = result + sum.
And then n = n / 10;
Finally, Ouside the while loop to check the input is equal to result, if it's true
-log as it is peterson number

->code
let input = 145;
let n = input;
let remainder;
let result = 0;
while(n > 0)
{
    remainder = n % 10;
    let sum = 1;
    for(let i=1; i<= remainder; i++)
    {
        sum = sum * i;
    }
    result = result + sum;
    n = Math.floor(n/10);
}
if(result === input)
{
    console.log("peterson Number");
}else
{
    console.log("Not a peterson Number")
}

5) Binary search by recursive:
Step 1: Array should be ascending order
Algorithm:
A = array, l = low, h = high, n = value to be search
RBinarySearch(A,l,h,n)
{
    if(l == h)
    {
        if(A[l] == v)
        {
            return l
        }else{
            return 0;
        }
    }else{
        mid = (low + high)/2
        if(A[mid] == n)
        {
            return mid;
        }else if(A[mid] < n)
        {
            return RBinarySearch(A, l, mid-1,n)
        }else
        {
            return RBinarySearch(A,mid+1,h,n)
        }
    }
}

{
    //prime number
    num++;

            break;

    let temp = num; sum = 0, mul = 1;

        rem = temp % 10;

        mul *= rem;
        temp = Math.floor(temp/10);

    let square = num* num;
    while(tempAt > 0)
    {
        digitCount++;
        tempAt = Math.floor(tempAt/10);

    let remainder = square % Math.pow(10,digitCount);

    {
        digitcountArms++;
        index = Math.floor(index/10)

    }
    //peterson number
    let tempPet = num, remPet, result = 0;

    {
        remPet = tempPet % 10;
        let sum = 1;
        for(let i=1; i<= remPet; i++)
        {
            sum = sum * i;
        }
        result = result + sum;
        tempPet = Math.floor(tempPet/10);

    if(i === num && sum === mul && remainder === num && sumArms == num && result === num)

static int secondLargest(int[] arr) // [5, 2, 5, 8, 9, 6]
{
    int largest = -1, sLargest = -1;

    for(int i=0; i< arr.length; i++)
    {
        if(arr[i] > largest)
        {
            sLargest = largest;
            largest = arr[i];
        }
        else if(arr[i] < largest && arr[i] > sLargest)
        {
            sLargest = arr[i];
        }
    }

    return sLargest;
}

function sum(ind, arr, target, res, curr) {
    if (target === 0) {
        res.push([...curr]); // If target is 0, we found a valid combination
        return;
    }

    for (let i = ind; i < arr.length; i++) {
        if (arr[i] > target) break; // If the current number is greater than the target, stop further processing

        // Skip duplicates to avoid duplicate combinations
        if (i > ind && arr[i] === arr[i - 1]) continue;

        curr.push(arr[i]); // Include the current number in the combination
        sum(i + 1, arr, target - arr[i], res, curr); // Recur with the next index and reduced target
        curr.pop(); // Backtrack and remove the last number to try other combinations
    }
}

function kadaneWithIndices(arr) {
  let maxSoFar = arr[0]; // Stores the maximum sum encountered so far
  let currentMax = arr[0]; // Stores the sum of the current subarray
  let start = 0, end = 0, tempStart = 0; // Variables to track indices

  for (let i = 1; i < arr.length; i++) {
    // If the current element alone is better than the current subarray, reset
    if (arr[i] > currentMax + arr[i]) {
      currentMax = arr[i];
      tempStart = i; // Update the starting index for the current subarray
    } else {
      currentMax += arr[i];
    }

    // Update the maximum sum and indices if a new maximum is found
    if (currentMax > maxSoFar) {
      maxSoFar = currentMax;
      start = tempStart; // Finalize the start index
      end = i;           // Finalize the end index
    }
  }

  return {
    maxSum: maxSoFar,
    subarray: arr.slice(start, end + 1), // Extract the subarray
    indices: [start, end] // Indices of the subarray
  };
}

function isValid(col, row, arr) {
    let n = arr.length;

    // Check same row
    for (let i = 0; i < col; i++) {
        if (arr[row][i] == "Q") {
            return false;
        }
    }

    // Check upper diagonal
    for (let i = row - 1, j = col - 1; i >= 0 && j >= 0; i--, j--) {

            return false;
        }

            return false;
        }
    }

    return true;

var spiralOrder = function(matrix) {
    let temp = [];
    if (matrix.length === 0) return temp; // Handle empty matrix case

    let m = matrix.length;    // Number of rows
    let n = matrix[0].length; // Number of columns
    let top = 0;
    let bottom = m - 1;
    let left = 0;
    let right = n - 1;

    while (top <= bottom && left <= right) {
        // Traverse from left to right along the top row
        for (let i = left; i <= right; i++) {
            temp.push(matrix[top][i]);
        }
        top++;

        // Traverse from top to bottom along the right column
        for (let i = top; i <= bottom; i++) {
            temp.push(matrix[i][right]);
        }
        right--;

        // Traverse from right to left along the bottom row (if it exists)
        if (top <= bottom) {
            for (let i = right; i >= left; i--) {
                temp.push(matrix[bottom][i]);
            }
            bottom--;
        }

        // Traverse from bottom to top along the left column (if it exists)
        if (left <= right) {
            for (let i = bottom; i >= top; i--) {
                temp.push(matrix[i][left]);
            }
            left++;
        }
    }

    return temp;
};

class Solution:
    def generateTrees(self, n: int) -> List[Optional[TreeNode]]:

        arr = []
        for i in range(1, n+1):
            arr.append(i)

        result = self.permutations(arr)

        print(result)

        total = []
        seen = set()

        for i in result:
            root = None
            for j in i:
                root = self.BST(root, j)

            serial = self.serialize(root)
            if serial not in seen:
                seen.add(serial)
                total.append(root)

        return total


    def serialize(self, root):
        if not root:
            return "None"
        return f"{root.val},{self.serialize(root.left)},{self.serialize(root.right)}"


    def BST(self, root, val):

        if root is None:
            root = TreeNode(val)
            return root

        if val < root.val:
            root.left = self.BST(root.left, val)
        else:
            root.right = self.BST(root.right, val)

        return root


    def permutations(self, arr):
        n = len(arr)
        if n <= 1:
            return [arr]

        result = []

        for i in range(n):
            current = arr[i]
            rem = arr[:i] + arr[i+1:]

            for p in self.permutations(rem):
                result.append([current] + p)

        return result

Given an integer n, return the number of structurally unique BST's
(binary search trees) which has exactly n nodes of unique values from 1 to n.


class Solution:
    def numTrees(self, n: int) -> int:
        # Using Catalan number formula: C(2n,n)/(n+1)
        result = 1
        for i in range(n):
            result = result * (2 * n - i) // (i + 1)
        return result // (n + 1)
